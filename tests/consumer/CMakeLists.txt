cmake_minimum_required(VERSION 3.24)
project(toolchain_consumer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Threads REQUIRED)
find_package(fmt CONFIG REQUIRED PATHS "${TOOLCHAIN_PREFIX}/lib/cmake/fmt" NO_DEFAULT_PATH)
find_package(spdlog CONFIG REQUIRED PATHS "${TOOLCHAIN_PREFIX}/lib/cmake/spdlog" NO_DEFAULT_PATH)
find_package(Protobuf CONFIG REQUIRED PATHS "${TOOLCHAIN_PREFIX}/lib/cmake/protobuf" NO_DEFAULT_PATH)
find_package(re2 CONFIG REQUIRED PATHS "${TOOLCHAIN_PREFIX}/lib/cmake/re2" NO_DEFAULT_PATH)
find_package(antlr4-runtime CONFIG REQUIRED PATHS "${TOOLCHAIN_PREFIX}/lib/cmake/antlr4-runtime" NO_DEFAULT_PATH)
add_executable(toolchain_consumer main.cpp antlr.cpp)
target_include_directories(toolchain_consumer PRIVATE "${TOOLCHAIN_PREFIX}/include" "${TOOLCHAIN_PREFIX}/include/antlr4-runtime")
target_link_libraries(toolchain_consumer PRIVATE protobuf::libprotobuf re2::re2 antlr4_static fmt::fmt spdlog::spdlog Threads::Threads)
foreach(name z zstd lz4 lzma bz2 modbus gflags)
  find_library(TEST_${name} NAMES ${name} PATHS "${TOOLCHAIN_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(toolchain_consumer PRIVATE ${TEST_${name}})
endforeach()
enable_testing()
add_test(NAME toolchain_consumer COMMAND toolchain_consumer)
